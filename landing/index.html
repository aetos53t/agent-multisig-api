// flow

How it works

01

Register agents

Each agent registers pubkey + webhook. Private keys never leave their wallet.

02

Create vault

Combine agents into m-of-n vault. We derive shared addresses per chain.

03

Propose spend

Any agent proposes. We build the signing payload and notify others via webhook.

04

Collect signatures

Agents sign asynchronously with their own wallets. Any provider works.

05

Execute

Threshold met → finalize transaction → broadcast to network.

// endpoints

API overview

Method Endpoint Description
POST /agents Register agent with pubkey + webhook
GET /agents/{id} Get agent details
POST /multisigs Create m-of-n vault from agents
GET /multisigs/{id} Get vault with derived addresses
POST /proposals Create spending proposal
POST /proposals/{id}/signatures Submit signature from agent
GET /proposals/{id} Get proposal status + signatures

webhook-handler.ts 
// Your agent receives a webhook when signature is needed
export async function handleWebhook(req: Request) {
  const { proposalId, chain, digest } = await req.json();
  
  // Sign with your wallet (any provider)
  const signature = await myWallet.signDigest(digest);
  
  // Submit back to coordination API
  await fetch(`/proposals/${proposalId}/signatures`, {
    method: 'POST',
    body: JSON.stringify({
      agentId: MY_AGENT_ID,
      signature
    })
  });
  
  // That's it. Works for BTC, STX, ETH...
}
